    Public ReadOnly Property IsDisassemblyPart() As Boolean
        Get
            If (ultraGridERPPurchaseOrders Is Nothing) Then
                Throw New Exception("Unable to load the Infragistics package. =(")
            End If
            If (ultraGridERPPurchaseOrders.ActiveRow IsNot Nothing) Then
                If (ultraGridERPPurchaseOrders.ActiveRow.HasParent()) Then
                    Dim PoLine = TryCast(ultraGridERPPurchaseOrders.ActiveRow.ParentRow.ListObject, QuickReportSchema)
                    Dim child = TryCast(ultraGridERPPurchaseOrders.ActiveRow.ListObject, DisassemblyPart)
                    If (PoLine.PartNo = child.ChildPartNo) Then
                        Return False
                    End If
                    Return True
                End If
            End If
            Return False
        End Get
    End Property


    Public ReadOnly Property IsSelfContainedDisassembly() As Boolean
        Get
            If (ultraGridERPPurchaseOrders Is Nothing) Then
                Throw New Exception("Unable to load the Infragistics package. =(")
            End If
            If (ultraGridERPPurchaseOrders.ActiveRow IsNot Nothing AndAlso ultraGridERPPurchaseOrders.ActiveRow.HasParent()) Then
                Dim PoLine = TryCast(ultraGridERPPurchaseOrders.ActiveRow.ParentRow.ListObject, QuickReportSchema)
                Dim child = PoLine.Disassemblies.FirstOrDefault(Function(item) item.ChildPartNo = PoLine.PartNo)
                Return child IsNot Nothing
            End If

            Return False
        End Get
    End Property

    Public ReadOnly Property IsValidEntry() As (Boolean, String)
        Get
            Dim isValid = True
            Dim msg As String

            Dim po = _selected_ifs_po
            Dim selectedRow
            If (ultraGridERPPurchaseOrders.ActiveRow IsNot Nothing) Then
                selectedRow = TryCast(ultraGridERPPurchaseOrders.ActiveRow.ListObject, QuickReportSchema)
            End If


            If (po Is Nothing AndAlso selectedRow IsNot Nothing) Then
                po = selectedRow
            End If

            If (cboRef4.SelectedValue = "PURCADD" And Not IsDisassemblyPart) Then
                If (_selected_ifs_po Is Nothing) Then
                    _selected_ifs_po = po
                End If

                If ((po Is Nothing OrElse ultraGridERPPurchaseOrders.ActiveRow Is Nothing) OrElse po.PartNo <> txtPartNo.Text OrElse po.OrderNo <> txtPONumber.Text) Then
                    isValid = False
                    msg = "Is Mandatory to select a PO Line in order to add items to inventory."
                    'ElseIf (po IsNot Nothing AndAlso CurrentFinesseQty > po.InvQtyArrived) Then
                    '    isValid = False
                    '    msg = "Overstocking inventory is not allowed. The quantity added to the inventory cannot exceed the quantity specified in the Purchase Order."
                ElseIf (po IsNot Nothing AndAlso Not po.IsValidSchema) Then
                    isValid = False
                    msg = po.SchemaError
                End If
            End If
            If (_userPermission = UserAccess.FULL_ACCESS AndAlso Not isValid) Then
                Dim result = MessageBox.Show($"The supplied PO information is not valid to be Proccessed. {vbNewLine}This won't send a succesfull message to IFS.{vbNewLine}Are you sure you want to continue?", "Info", MessageBoxButtons.YesNo, MessageBoxIcon.Information)
                If (result = DialogResult.Yes) Then
                    Return (True, "FULL_ACCESS")
                End If
            End If

            Return (isValid, msg)
        End Get

    Private Sub SendIFSToFinesseMessage()
        If clickedRow IsNot Nothing Then
            Dim row = TryCast(clickedRow.ListObject, QuickReportSchema)

            Dim qty = 0
            Integer.TryParse(txtQuantityMessage.Text, qty)
            If (String.IsNullOrEmpty(txtQuantityMessage.Text) OrElse qty = 0) Then
                Dim toast As New ToastNotification($"You need to specify a valid quantity!", Color.Red)
                toast.Show()
                Exit Sub
            End If

            Dim result = row.SendMessage(_dtCurrency, txtQuantityMessage.Text)
            If (Not String.IsNullOrEmpty(result)) Then
                Dim toast As New ToastNotification($"Message successfully Sent! [{result}]", Color.Green)
                toast.Show()
            Else
                Dim toast As New ToastNotification($"Error sending the message!", Color.Red)
                toast.Show()
            End If
            UltraGridMenuStrip.Close()
        End If
    End Sub

    Private Sub ReloadPermissions()
        If (UserPermission = UserAccess.FULL_ACCESS OrElse
                    (UserPermission <> UserAccess.NO_ACCESS AndAlso cboRef4.SelectedValue <> "PURCADD") OrElse
                    (UserPermission = UserAccess.PO_MANDATORY AndAlso _selected_ifs_po IsNot Nothing) OrElse
                    (UserPermission = UserAccess.RECIEVED_PO_MANDATORY AndAlso _selected_ifs_po IsNot Nothing AndAlso _selected_ifs_po.IsRecieved)) Then
            SaveToolStripMenuItem.Enabled = True
            btnCommit.Enabled = True
            btnAddItemToList.Enabled = True
            txtPONumber.Enabled = cboRef4.SelectedValue = "MANFADD" OrElse (UserPermission = UserAccess.FULL_ACCESS)
            LblPermissionMsg.Text = ""
            If (cboRef4.SelectedValue <> "PURCADD" OrElse (_selected_ifs_po IsNot Nothing AndAlso txtPartNo.Text <> _selected_ifs_po.PartNo)) Then
                _selected_ifs_po = Nothing
                txtPONumber.Text = ""
                txtQty.Text = ""
            ElseIf (_selected_ifs_po IsNot Nothing) Then
                txtPONumber.Text = _selected_ifs_po.OrderNo
            End If
        Else
            SaveToolStripMenuItem.Enabled = False
            btnCommit.Enabled = False
            btnAddItemToList.Enabled = False
            txtPONumber.Enabled = False

            If (cboRef4.SelectedValue = "PURCADD") Then
                If ((_selected_ifs_po Is Nothing OrElse ultraGridERPPurchaseOrders.ActiveRow Is Nothing) OrElse _selected_ifs_po IsNot Nothing AndAlso _selected_ifs_po.PartNo <> txtPartNo.Text) Then
                    LblPermissionMsg.Text = "You need to select a PO"
                Else

                    If ({UserAccess.NO_ACCESS, UserAccess.GENERAL_USER}.Contains(UserPermission) OrElse
                    UserPermission <> UserAccess.FULL_ACCESS AndAlso _selected_ifs_po Is Nothing OrElse
                    UserPermission = UserAccess.RECIEVED_PO_MANDATORY AndAlso _selected_ifs_po IsNot Nothing AndAlso Not _selected_ifs_po.IsRecieved) Then
                        LblPermissionMsg.Text = "You don't have permission to perform this operation"

                    End If
                End If
            End If

            '    If ((cboRef4.SelectedValue = "PURCADD" AndAlso Not {UserAccess.NO_ACCESS, UserAccess.GENERAL_USER}.Contains(UserPermission) AndAlso Not String.IsNullOrEmpty(txtPartNo.Text)) _
            '        AndAlso (UserPermission = UserAccess.RECIEVED_PO_MANDATORY AndAlso _selected_ifs_po IsNot Nothing AndAlso _selected_ifs_po.IsRecieved) _
            '        OrElse _selected_ifs_po IsNot Nothing AndAlso _selected_ifs_po.PartNo <> txtPartNo.Text) Then
            '    LblPermissionMsg.Text = "You need to select a PO"
            'Else

            'End If

    Private Sub frmAddInventory_FormClosing(ByVal sender As Object, ByVal e As FormClosingEventArgs) Handles Me.FormClosing
        Dim sw As New Stopwatch
        sw.Start()

        Debug.Print(sw.ElapsedMilliseconds & " Before DoYouWantToSave")

        Dim lShouldReturn As Boolean
        DoYouWantToSave(lShouldReturn)
        If lShouldReturn Then Return

        Debug.Print(sw.ElapsedMilliseconds & " After DoYouWantToSave")

        Try
            Debug.Print(sw.ElapsedMilliseconds & " Before OpenConnection")

            Dim newConn As New SqlConnection(FinesseConnectionString)
            newConn.Open()

            Debug.Print(sw.ElapsedMilliseconds & " After OpenConnection")

            Dim SettingsToSave As New List(Of AppConfigState) From {
                New AppConfigState() With {.Cmndsel = cmndsel, .Tag = "MainFormSize", .Rvalue = Me.Size.ToString},
                New AppConfigState() With {.Cmndsel = cmndsel, .Tag = "MainFormLocation", .Rvalue = Me.Location.ToString},
                New AppConfigState() With {.Cmndsel = cmndsel, .Tag = "MainFormWindowState", .Rvalue = Me.WindowState.ToString}
            }

            If (Not String.IsNullOrEmpty(cboOwner.SelectedValue)) Then
                SettingsToSave.Add(New AppConfigState() With {.Cmndsel = cmndsel, .Tag = "LastOwnerCode", .Rvalue = cboOwner.SelectedValue})
            End If

            If (Not String.IsNullOrEmpty(cboWarehouse.SelectedValue)) Then
                SettingsToSave.Add(New AppConfigState() With {.Cmndsel = cmndsel, .Tag = "LastSelectedWarehouse", .Rvalue = cboWarehouse.SelectedValue})
            End If

            Debug.Print(sw.ElapsedMilliseconds & " After array of settings")

            SettingsToSave.SaveConfig()

            newConn.Close()
            newConn.Dispose()

            Debug.Print(sw.ElapsedMilliseconds & " After Close Connection")

        Catch ex As Exception
            Console.WriteLine("Error saving App Configs: ")
            Console.WriteLine(ex.Message)
        End Try

    End Sub

    Private Sub RefreshCurrentInventory()
        Dim currentTotal As String
        Dim partno As String = txtPartNo.Text
        Dim owner As String = cboOwner.SelectedValue
        Dim warehouse As String = cboWarehouse.SelectedValue

        If String.IsNullOrEmpty(partno) OrElse String.IsNullOrEmpty(owner) OrElse String.IsNullOrEmpty(warehouse) Then
            lblCurrentInventoryWarning.Visible = True
            Exit Sub
        Else
            lblCurrentInventoryWarning.Visible = False
        End If

        Dim sSQL As New StringBuilder
        sSQL.AppendLine("select Qty = ps.onhand, Barcode = ps.unique_no, [Phase No] = ps.batchno, [Phase Desc] = g.entitydesc, WH = ps.bld, [Owner] = ps.dept, [Return] = pj.todate, ps.receipt")
        sSQL.AppendLine("from dbo.inpartsub ps")
        sSQL.AppendLine("join dbo.glentities g on ps.batchno = g.entityno")
        sSQL.AppendLine("left outer join dbo.pjjobbudexp pj on g.entityno = pj.entityno and ps.partno = pj.partno")
        sSQL.AppendLine("where ps.partno = " & partno.SQLQuote)
        sSQL.AppendLine("and ps.dept = " & owner.SQLQuote)
        sSQL.AppendLine("and ps.bld = " & warehouse.SQLQuote)
        sSQL.AppendLine("and ps.onhand > 0")
        sSQL.AppendLine("order by ps.receipt DESC, ps.unique_no")

        Using newConn = New SqlConnection(FinesseConnectionString)
            Dim t = newConn.GetDataTable(sSQL)
            dgvCurrentInventory.DataSource = t
            currentTotal = IIf(String.IsNullOrEmpty(t.Compute("Sum(qty)", "").ToString), "0", t.Compute("Sum(qty)", "").ToString)
            newConn.Close()
        End Using

        tsslCurrentInventoryTotalQty.Text = $"Total Qty: {currentTotal}"
        ItemsForImportQtyUpdating()
    End Sub

    Private Sub BulkSave(ByVal tSource As DataTable, ByVal destinationTable As String)
        Dim tTemp As New DataTable

        tTemp.Columns.Add("is_insert", GetType(Boolean)).DefaultValue = False
        tTemp.Columns.Add("is_update", GetType(Boolean)).DefaultValue = False
        tTemp.Columns.Add("is_delete", GetType(Boolean)).DefaultValue = False

        For Each c As DataColumn In tSource.Columns
            If Not c.ReadOnly Then tTemp.Columns.Add(c.ColumnName, c.DataType) ' assuming that read-only columns will by calculated automatically by the DB
        Next

        For Each r As DataRow In tSource.Rows
            Select Case r.RowState
                Case DataRowState.Added, DataRowState.Deleted, DataRowState.Modified
                    tTemp.ImportRow(r)
                Case Else
                    Debug.Assert(r.RowState = DataRowState.Unchanged)
            End Select
        Next

        For Each r As DataRow In tTemp.Rows
            Select Case r.RowState
                Case DataRowState.Added
                    r("is_insert") = True
                Case DataRowState.Modified
                    r("is_update") = True
                Case DataRowState.Deleted
                    r.RejectChanges() ' necessary so that we can look at and modify fields.
                    r("is_delete") = True
            End Select
        Next

        Using bulkCopy As New SqlBulkCopy(gConn)
            bulkCopy.DestinationTableName = destinationTable
            bulkCopy.BatchSize = 0
            bulkCopy.ColumnMappings.Clear()

            For Each c As DataColumn In tTemp.Columns
                Debug.Print(c.ColumnName)
                bulkCopy.ColumnMappings.Add(c.ColumnName, c.ColumnName)
            Next

            bulkCopy.WriteToServer(tTemp)
        End Using
    End Sub

    Private Async Sub btnAddItemToList_Click(ByVal sender As Object, ByVal e As EventArgs) Handles btnAddItemToList.Click
        Dim Validation = IsValidEntry
        If (Not Validation.Item1) Then
            MsgBox(Validation.Item2, MsgBoxStyle.Critical)
            Return
        End If

        If (String.IsNullOrEmpty(txtSerialNo.Text) And String.IsNullOrEmpty(txtBarcode.Text) And String.IsNullOrEmpty(txtQty.Text)) Then
            txtBarcode.Focus()
            Return
        End If

        If Not String.IsNullOrEmpty(txtSerialNo.Text) And String.IsNullOrEmpty(txtBarcode.Text) Then
            txtBarcode.Focus()
            Return
        End If

        Dim qty As Integer

        Try
            qty = CInt(txtQty.Text)
        Catch ex As Exception
            MsgBox("Quantity must be a whole number greater than zero.", vbOKOnly)
            Return
        End Try

        If CInt(txtQty.Text) < 1 Then
            MsgBox("Quantity must be a whole number greater than zero.", vbOKOnly)
            Return
        End If

        Dim lShouldReturn As Boolean
        ValidateNewInventoryItemInfo(lShouldReturn)
        If lShouldReturn Then Return

        'Set ShouldLinkChildParts to false as the default before any "add item" processing begins.
        Dim ShouldLinkChildParts As Boolean = False
        Dim dtLinkedParts As New DataTable

        Try
            Dim childPartLinkingAllowed As Boolean = DirectCast(cboOwner.SelectedItem, DataRowView)("EnableInpartsubRefChildLinking")

            If Not childPartLinkingAllowed Then GoTo StagePartForAdd


            Using conn As New SqlConnection(FinesseConnectionString), cmd As New SqlCommand("GetLinkedChildPartInformation", conn), ad As New SqlDataAdapter(cmd)
                conn.Open()
                cmd.CommandType = CommandType.StoredProcedure
                With cmd.Parameters
                    ''Set the size of VarChar max to -1
                    .Add("@partno", SqlDbType.VarChar, 50).Value = txtPartNo.Text
                    .Add("@qty", SqlDbType.Int).Value = qty
                    .Add("@bld", SqlDbType.VarChar, 10).Value = cboWarehouse.SelectedValue
                    .Add("@owner", SqlDbType.VarChar, 10).Value = cboOwner.SelectedValue
                End With
                ad.Fill(dtLinkedParts)
            End Using


            'Child Part Count
            If dtLinkedParts IsNot Nothing AndAlso dtLinkedParts.Rows.Count > 0 Then
                Dim msg As New StringBuilder
                msg.AppendLine("Parent part " & txtPartNo.Text & ": " & txtPartDescEngOp.Text & " has linked parts associated with it.  By adding this part, you will be removing linked parts from working inventory:")
                msg.AppendLine("") 'Add an empty space between the message and the parts
                For Each r In dtLinkedParts.Rows
                    msg.AppendLine(r("childPartQtyToLink") & " of " & r("childPartno") & ": " & r("childPartDesc"))
                Next
                msg.AppendLine("") 'Add an empty space between the parts and the working inventory question.
                msg.AppendLine("Do you want to remove linked parts from working inventory?")


                Dim result = MsgBox(msg.ToString, MsgBoxStyle.YesNoCancel, "Linked Parts")

                Select Case result
                    Case MsgBoxResult.Yes
                        Dim childPartsGoingNegativeQty = From r In dtLinkedParts Where (r("childOnhand") - r("childPartQtyToLink")) < 0
                        Dim errExists As Boolean = False
                        For Each r In childPartsGoingNegativeQty
                            'errMsg.AppendLine("There is not enough [Part Type] [Part Description] in [Warehouse] owned by [Owner] to be linked to [Parent Part Description].  The [Parent Part Description] will not be added to inventory.")

" & r("childPartQtyToLink") & " " & IsAre(r("childPartQtyToLink")) & " required to be linked to this part." & vbCrLf & vbCrLf & "The " & txtPartDescEngOp.Text & " will not be added to inventory."

                        ShouldLinkChildParts = True
                    Case MsgBoxResult.No
                        ShouldLinkChildParts = False
                    Case MsgBoxResult.Cancel
                        'Quit, do nothing
                        Exit Sub
                End Select
            End If

            Dim IsValidPO As Boolean = cboRef4.SelectedValue = "PURCADD" AndAlso _selected_ifs_po IsNot Nothing AndAlso _selected_ifs_po.PartNo = txtPartNo.Text AndAlso _selected_ifs_po.OrderNo = txtPONumber.Text

StagePartForAdd:

                    Dim totalItemsAddedQty = t.Sum(Function(r) r)
                    Dim onhand = FinesseInventory.onhand + CInt(txtQty.Text) + totalItemsAddedQty
                    If (onhand > maxqty) Then

    Private Sub AddItemRow(ByVal ShouldLinkChildParts As Boolean, ByVal POValidation As (Boolean, String))
        Dim ipAddress = Nothing

        If chkAutoAssignIPAddress.Checked Then
            ipAddress = getNextAvailableIPAddress(txtPartNo.Text)
        End If


        Dim batchno As String
        Dim internalCustomerResult As (String, String) 'Result, Error
        Dim ActiveRow = ultraGridERPPurchaseOrders.ActiveRow

        Dim isIFSPOLine = _selected_ifs_po IsNot Nothing AndAlso ActiveRow IsNot Nothing AndAlso cboRef4.SelectedValue = "PURCADD" AndAlso _selected_ifs_po.PartNo = txtPartNo.Text AndAlso _selected_ifs_po.OrderNo = txtPONumber.Text


        Dim _partno = txtPartNo.Text
        Dim _partdesc = txtPartDescEngOp.Text
        Dim _poPartNo = _partno

        Dim parentPartNo As String

        Dim POToWorkWith As QuickReportSchema = _selected_ifs_po
        Dim parent = Nothing

        If (ActiveRow IsNot Nothing AndAlso ActiveRow.HasParent) Then
            parent = TryCast(ActiveRow.ParentRow.ListObject, QuickReportSchema)
        End If

        If (IsDisassemblyPart) Then
            parentPartNo = parent.PartNo
            _poPartNo = parent.PartNo
            If Not IsSelfContainedDisassembly Then
                POToWorkWith = parent
                _poPartNo = parent.PartNo
            End If
        End If

        If (Not IsDisassemblyPart AndAlso parent IsNot Nothing) Then
            POToWorkWith = parent
            _poPartNo = parent.PartNo
        End If


        If (isIFSPOLine AndAlso (Not IsDisassemblyPart Or Not IsSelfContainedDisassembly)) Then
            batchno = POToWorkWith.GetBatchNo(txtSerialNo.Text)
            internalCustomerResult = POToWorkWith.GetInternalCustomer(_dtCurrency)
            Console.WriteLine($"Line Internal Customer: {internalCustomerResult.Item1}")
            If (Not IsDisassemblyPart) Then
                If (internalCustomerResult.Item1 Is Nothing) Then
                    MsgBox(internalCustomerResult.Item2, MsgBoxStyle.Exclamation)
                End If
                If (Not POToWorkWith.IsValidSchema) Then
                    MsgBox(POToWorkWith.SchemaError, MsgBoxStyle.Exclamation)

    Private Sub hideOrDisplayAutoIPAddressForPart(partno As String)

        Dim controls = {Label5, txtSerialNo, pb2Dbarcode, lblBarcode, txtBarcode, btnGetNextBarcode, btnPrintSingleBarcode, lblQty, txtQty}

        If checkIfPartHasIPRangeSet(partno) Then

                    c.Top = c.Top + txtMACAddress.Height
                Next
            End If



            chkAutoAssignIPAddress.Checked = True
            chkAutoAssignIPAddress.Visible = True
            lblMACAddress.Visible = True
            txtMACAddress.Visible = True
            chkDeferEnteringMacAddress.Visible = True
            chkDeferEnteringMacAddress.Checked = False


        Else

            If txtMACAddress.Visible Then
                For Each c As Control In controls
                    c.Top = c.Top - txtMACAddress.Height
                Next
            End If

            chkAutoAssignIPAddress.Visible = False
            lblMACAddress.Visible = False

            txtMACAddress.Text = String.Empty

            'force focus to the serial number after entering an item
            chkDeferEnteringMacAddress.Checked = True

    Private Function checkIfPartHasIPRangeSet(partno As String) As Boolean
        Dim conn As SqlConnection = GetOpenedFinesseConnection()
        Dim isRangeSet As Boolean

        isRangeSet = conn.ExecuteScalar($"SELECT CASE WHEN ip.RangeType IS NULL THEN 0
                                          	   ELSE 1 END as isSet
                                          FROM dbo.inpart ip
                                          WHERE ip.partno = '{partno}'")
        conn.Close()
        conn.Dispose()

        Return isRangeSet
    End Function

    Private Function getNextAvailableIPAddress(partno As String) As String

        Dim IPAddress As String

        Dim sqlComm As SqlCommand = New SqlCommand("dbo.Get_next_available_IPAddress", gConn)
        sqlComm.CommandType = CommandType.StoredProcedure
        sqlComm.Parameters.Add("@partno", SqlDbType.VarChar, 50).Value = txtPartNo.Text
        sqlComm.Parameters.Add("@nextIPAddress", SqlDbType.VarChar, 15)
        sqlComm.Parameters("@nextIPAddress").Direction = ParameterDirection.Output
        sqlComm.ExecuteNonQuery()
        IPAddress = sqlComm.Parameters.Item("@nextIPAddress").Value

        Return IPAddress
    End Function

                                             " and bld=" & SQLQuote(cboWarehouse.SelectedValue) &

    Private Sub ReloadSavedPOs(refreshedData As DataTable)
        Try

            Me.Cursor = Cursors.WaitCursor
            If (refreshedData IsNot Nothing AndAlso refreshedData.Rows.Count > 0) Then
                For Each item As DataRow In refreshedData.Rows
                    Dim newQty
                    Dim dataResult As DataTable
                    Using conn = GetOpenedFinesseConnection()
                        Dim cmd = conn.CreateCommand
                        cmd.CommandText = "SELECT TOP(1) OrderNo, PartNumber, Qty, Line_No, Release_No 
                                           FROM IFS.InventoryReceipts
                                           WHERE OrderNo = @OrderNo
                                             AND PartNumber = @PartNumber
                                             AND Line_No = @LineNo
                                             AND Release_No = @ReleaseNo"

                        cmd.Parameters.Clear()
                        cmd.Parameters.AddWithValue("@OrderNo", item("PONumber"))
                        cmd.Parameters.AddWithValue("@PartNumber", item("partno"))
                        cmd.Parameters.AddWithValue("@LineNo", item("POLineNo"))
                        cmd.Parameters.AddWithValue("@ReleaseNo", item("POReleaseNo"))
                        Using reader = cmd.ExecuteReader()
                            If reader.Read() Then
                                If Not reader.IsDBNull(reader.GetOrdinal("Qty")) Then
                                    newQty = Integer.Parse(reader("Qty"))
                                End If
                            End If
                        End Using
                    End Using

                    Dim element = _ifs_data.FirstOrDefault(Function(po) po.PartNo = item("partno") AndAlso po.OrderNo = item("PONumber") AndAlso po.LineNo = item("POLineNo") AndAlso po.ReleaseNo = item("POReleaseNo"))
                    If (element IsNot Nothing AndAlso newQty IsNot Nothing) Then
                        element.FinessePart.QtyInFinesse = newQty
                    End If
                Next
                Dim selectedRow = GetParentchildSelectedRow()

                ultraGridERPPurchaseOrders.DataSource = _ifs_data.ToList()
                ultraGridERPPurchaseOrders.Selected.Rows.Clear()
                ultraGridERPPurchaseOrders.ActiveRow = Nothing

                filter_ERP_DataGrid(txtERPPoNumber.Text)
                SelectPreviousSelectionOnGrid(selectedRow.Item1, selectedRow.Item2)
            End If
        Catch ex As Exception
            Me.Cursor = Cursors.Default
            Console.WriteLine(ex.Message)
            Dim toast As New ToastNotification("Error loading saved PO Data to the grid", Color.Red)
            toast.Show()
        Finally
            Me.Cursor = Cursors.Default
        End Try
    End Sub

    Private Sub ultraGridERPPurchaseOrders_AfterRowActivate(sender As Object, e As EventArgs) Handles ultraGridERPPurchaseOrders.AfterRowActivate
        Try
            If (IsGridLoading) Then
                Exit Sub
            End If

            Dim selectedRow As UltraGridRow = ultraGridERPPurchaseOrders.ActiveRow
            Dim PoLine As QuickReportSchema
            If (Not selectedRow.HasParent) Then
                PoLine = TryCast(selectedRow.ListObject, QuickReportSchema)
                Dim childBand = selectedRow.ChildBands(0)
                If childBand.Rows.Count > 0 Then
                    childBand.Rows(0).Activate()
                    Exit Sub
                End If
            Else
                Dim disassembly = TryCast(selectedRow.ListObject, DisassemblyPart)
                Dim parent = TryCast(selectedRow.ParentRow.ListObject, QuickReportSchema).Clone()

                If (parent.PartNo <> disassembly.ChildPartNo) Then
                    parent.PartNo = disassembly.ChildPartNo
                    parent.IFSInventory = parent.IFSInventory.Where(Function(item) item.PartNo = disassembly.ChildPartNo).ToList()
                    parent.InvQtyArrived = disassembly.Quantity
                    'parent.FinesseQty = 0
                End If

                If (parent IsNot Nothing AndAlso Not parent.IsSerializedPo) Then
                    txtQty.Text = disassembly.Quantity
                Else
                    txtQty.Text = 1
                End If
                PoLine = parent
            End If
            _selected_ifs_po = PoLine
            If (PoLine IsNot Nothing) Then
                cboRef4.SelectedValue = "PURCADD"
                If (Not String.IsNullOrEmpty(PoLine.Owner)) Then
                    cboOwner.SelectedValue = PoLine.Owner
                End If
                If (Not String.IsNullOrEmpty(PoLine.Warehouse)) Then
                    cboWarehouse.SelectedValue = PoLine.Warehouse
                End If

                txtPONumber.Text = PoLine.OrderNo
                txtPartNo.Text = PoLine.PartNo
                hideOrDisplayAutoIPAddressForPart(PoLine.PartNo)
                ReloadPermissions()
            End If

        Catch ex As Exception
            Console.WriteLine("Error Activating ultragrid row...")
            Console.WriteLine(ex.Message)
        End Try
    End Sub

    Private Function GetErpBaseData() As IEnumerable(Of QuickReportSchema)
        If (_ifs_data IsNot Nothing AndAlso cboErpWarehouse.SelectedItem IsNot Nothing) Then
            Dim selectedCompany = cboErpWarehouse.SelectedItem.Row("IFSCompanyCd").ToString
            Dim result = _ifs_data.Where(Function(po) _
                                        (String.IsNullOrEmpty(selectedCompany) OrElse String.IsNullOrEmpty(po.Contract) OrElse po.Contract = selectedCompany OrElse po.Warehouse = selectedCompany) AndAlso
    (String.IsNullOrEmpty(cboErpOwner.SelectedValue) OrElse String.IsNullOrEmpty(po.Owner) OrElse po.Owner = cboErpOwner.SelectedValue) AndAlso
                                        (chkUnexistingParts.Checked OrElse (Not chkUnexistingParts.Checked AndAlso (po.FinessePart IsNot Nothing OrElse (po.Disassemblies IsNot Nothing AndAlso po.Disassemblies.Count > 0)))) AndAlso
    (Not chkRecievedOnly.Checked OrElse
        (po.IsRecieved = True AndAlso
         CInt(po.FinesseQty) < CInt(po.BuyQtyDue) AndAlso 'Filtering by Qty ordered instead of qty arrived to show rows that haven't been fully receipt.
         (po.FinessePart IsNot Nothing OrElse po.Disassemblies IsNot Nothing)))
)



            Return result
        End If

        Return New List(Of QuickReportSchema)()
    End Function



    Private Sub filter_ERP_DataGrid(Optional search As String = Nothing)
        If (_ifs_data IsNot Nothing) Then
            Dim data = GetErpBaseData()
            Dim dataResult As IEnumerable(Of QuickReportSchema) = data
            If (Not String.IsNullOrEmpty(search)) Then
                Dim lowerSearchText = search.ToLower()

                Dim allWords = lowerSearchText.Split({" "}, StringSplitOptions.RemoveEmptyEntries)

                dataResult = data.Where(Function(po) _
                                        allWords.All(Function(word) _
                                            (po.PartNo IsNot Nothing AndAlso po.PartNo.ToLower().Contains(word)) OrElse
                                            (po.FinesseDescription IsNot Nothing AndAlso po.FinesseDescription.ToLower().Contains(word)) OrElse
                                            (po.OrderNo IsNot Nothing AndAlso po.OrderNo.ToLower().Contains(word)) OrElse
                                            (po.SupplierID IsNot Nothing AndAlso po.SupplierID.ToLower().Contains(word)) OrElse
                                            (po.SupplierName IsNot Nothing AndAlso po.SupplierName.ToLower().Contains(word)) OrElse
                                            (po.PoState IsNot Nothing AndAlso po.PoState.ToLower().Contains(word))
                                        )
                                    )

            End If
            ultraGridERPPurchaseOrders.DataSource = (dataResult.ToList())

            ultraGridERPPurchaseOrders.Selected.Rows.Clear()
            ultraGridERPPurchaseOrders.ActiveRow = Nothing

            ultraGridERPPurchaseOrders.Rows.ExpandAll(True)

            SelectPartOnErpGrid()
            ReloadPermissions()
        End If
    End Sub

    Function ValidateAddToInventoryItems()
        Dim dt = DataSetAddInventory.inpartsub
        Dim NBCSummary = From r As DataSetAddInventory.inpartsubRow In dt.Rows
                         Where r.RemoveNonCoded = True
                         Group r By partno = r.partno, partdesc = r.partdesc, owner = r.dept, wh = r.bld Into Group
                         Select partno, partdesc, owner, wh, totalonhand = Group.Sum(Function(r) r.onhand), count = Group.Count()
        Dim dtNBCSummary = NBCSummary.ToDataTable

        For Each dr As DataRow In dtNBCSummary.Rows
            Dim sSQL As New StringBuilder
            sSQL.AppendLine("select onhand from dbo.inpartsub ps where ps.partno = " & SQLQuote(dr.Item("partno")) & " and ps.dept = " & SQLQuote(dr.Item("owner")) & " and ps.bld = " & SQLQuote(dr.Item("wh")) & "  and ps.batchno = 'INSTOCK' and ps.unique_no = '' and ps.parentunique_no = '' and onhand > 0")
            Dim qtyNBCtoRemove = gConn.GetDataTable(sSQL)
            If qtyNBCtoRemove.Rows.Count = 0 OrElse qtyNBCtoRemove.Rows(0).Item("onhand") < dr.Item("totalonhand") Then
                Beep_Fail()
                MsgBox("You have intended to remove non-barcoded inventory when adding barcodes for " & dr.Item("partno") & ": " & dr.Item("partdesc") & ", but " & IIf(qtyNBCtoRemove.Rows.Count = 0 OrElse qtyNBCtoRemove.Rows(0).Item("onhand") = 0, "no", "not enough") & " non-barcoded items exist in INSTOCK (" & dr.Item("wh") & "), owned by " & dr.Item("owner") & "." & vbCrLf & vbCrLf & "Try to find a matching non-barcoded item in MIA or on Late Returns to check into your INSTOCK before adding this item, or uncheck the 'Remove Non-Coded' checkbox for this item to INCREASE inventory levels." & vbCrLf & vbCrLf & "Commit has been aborted.", vbOKOnly)
                Return False
            End If
        Next
        Return True
    End Function

    Function ItemsForImportQtyUpdating()
        Dim totalItemsAddedQty As Integer = 0
        Dim selectedItemsAddedQty As Integer = 0

        If dgvItemsForImport.Rows.Count <> 0 Then
            Dim t = From r As DataGridViewRow In dgvItemsForImport.Rows
                    Select total = r.DataBoundItem("onhand")
            totalItemsAddedQty = t.Sum(Function(r) r)

            Dim s = From r As DataGridViewRow In dgvItemsForImport.SelectedRows
                    Select total = r.DataBoundItem("onhand")
            selectedItemsAddedQty = s.Sum(Function(r) r)
        End If

        tsslItemsForImportTotalQty.Text = $"Total Qty: {totalItemsAddedQty}"
        tsslItemsForImportSelectedRowQty.Text = $"Selected Qty: {selectedItemsAddedQty}"
        Return True
    End Function

    Private Sub MACAddressValidationSurrogateFunction()

        If txtMACAddress.Text = String.Empty Then
            Exit Sub
        End If


        'Dim hasLowerCaseLetters = Regex.Match(txtMACAddress.Text, "[a-f]")

        'If hasLowerCaseLetters.Success Or txtMACAddress.Text.Length <> 17 Then
        '    attemptParseMACAddress()
        '    Exit Sub
        'End If


        If Not isValidMACAddress(txtMACAddress.Text) Then

            Dim parsedMACAddress As String = attemptParseMACAddress(txtMACAddress.Text)

            If Not isValidMACAddress(parsedMACAddress) Then
                Beep_Fail()
                MessageBox.Show("This is an invalid MAC address format. Please use the following format 12:34:56:78:9A:BC")
            Else
                txtMACAddress.Text = parsedMACAddress
            End If

        End If

    End Sub

    Private Function isValidMACAddress(MACAddress As String) As Boolean
        Return Regex.IsMatch(MACAddress, "^[0-9A-F]{2}:[0-9A-F]{2}:[0-9A-F]{2}:[0-9A-F]{2}:[0-9A-F]{2}:[0-9A-F]{2}$")
    End Function

    Private Function attemptParseMACAddress(MACAddress As String) As String
        Dim parsedMacAddress = txtMACAddress.Text.ToUpper

        'get rid of all separators
        parsedMacAddress = parsedMacAddress.Replace(".", "")
        parsedMacAddress = parsedMacAddress.Replace("-", "")
        parsedMacAddress = parsedMacAddress.Replace(":", "")
        parsedMacAddress = parsedMacAddress.Replace(" ", "")

        'if not 12 hexidecimal characters without separators then we cannot parse
        If parsedMacAddress.Length = 12 Then
            Dim i = parsedMacAddress.Length - 1
            While i > 0

                If i Mod 2 = 0 AndAlso i <> 0 Then
                    parsedMacAddress = parsedMacAddress.Insert(i, ":")
                End If

                i -= 1
            End While


        Else
            Return parsedMacAddress
        End If

        Return parsedMacAddress
    End Function

    Private Function getPartGUID(partno As String) As String
        If (String.IsNullOrEmpty(partno)) Then
            Return ""
        End If

        Dim partGUID As String = String.Empty
        Try
            Dim jammer As New SQLJammer(GetOpenedFinesseConnection())
            jammer.Add($"
                SELECT TOP 1 GUID FROM dbo.inpart
                WHERE partno =  {SQLQuote(partno)}",
                     Sub(result)
                         If result.Rows.Count > 0 Then
                             partGUID = result.Rows(0).Item("GUID").ToString()
                         End If
                     End Sub)
            jammer.Execute()
        Catch ex As Exception
            MessageBox.Show("Error Getting File Storage GUID: " & ex.Message)
        End Try
        Return partGUID
    End Function

    Private Function GetParentchildSelectedRow() As (QuickReportSchema, DisassemblyPart)
        Dim Selected = ultraGridERPPurchaseOrders.ActiveRow
        If (Selected Is Nothing) Then
            Return (Nothing, Nothing)
        End If
        Dim IsChild = Selected.HasParent
        Dim parent
        Dim child
        If (IsChild) Then
            parent = TryCast(Selected.ParentRow.ListObject, QuickReportSchema)
            child = TryCast(Selected.ListObject, DisassemblyPart)
        Else
            parent = TryCast(Selected.ListObject, QuickReportSchema)
        End If

        Return (parent, child)
    End Function

    Private Sub SelectPreviousSelectionOnGrid(parent As QuickReportSchema, child As DisassemblyPart)

                    If (childRow IsNot Nothing) Then
                        childRow.Activate()
                    End If
                End If
                row.Activate()
            End If
        End If
    End Sub
