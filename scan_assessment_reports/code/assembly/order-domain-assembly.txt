Private Sub ultraGridERPPurchaseOrders_MouseDown(sender As Object, e As MouseEventArgs) Handles ultraGridERPPurchaseOrders.MouseDown
        If e.Button = MouseButtons.Right Then
            Dim uiElement = ultraGridERPPurchaseOrders.DisplayLayout.UIElement.ElementFromPoint(New Point(e.X, e.Y))
            Dim row = TryCast(uiElement.GetContext(GetType(UltraGridRow)), UltraGridRow)


            If row IsNot Nothing Then
                clickedRow = row

                BtnAddSubparts.Visible = _can_add_edit_disassemblies_subparts
                UltraGridMenuStrip.Show(ultraGridERPPurchaseOrders, New Point(e.X, e.Y))
                menuItemSendMessage.Visible = _can_add_edit_disassemblies_subparts AndAlso po IsNot Nothing AndAlso po.IsRecieved

                txtQuantityMessage.Text = qty
            End If
        End If
    End Sub

Private Sub ultraGridERPPurchaseOrders_InitializeRow(sender As Object, e As Infragistics.Win.UltraWinGrid.InitializeRowEventArgs) Handles ultraGridERPPurchaseOrders.InitializeRow

        Dim row = e.Row
        row.Selected = False
        If row.Band.Index = 0 Then
            Dim PoLine = TryCast(row.ListObject, QuickReportSchema)
            If (PoLine.FinessePart Is Nothing) Then
                If (PoLine.Disassemblies Is Nothing OrElse PoLine.Disassemblies.Count = 0) Then
                    row.ToolTipText = "This part does not exist in Finesse, and no subparts have been added. Please create the part or add the required subparts."
                End If
                row.Appearance.BackColor = ColorPalette.DangerRow
            End If

            If (row.HasChild) Then
                row.ToolTipText = "This PO Line contains Subparts. Please receive all the subparts."
                row.Appearance.BackColor = ColorPalette.GrayRow
            End If
        End If

    End Sub

Private Sub ultraGridERPPurchaseOrders_AfterRowActivate(sender As Object, e As EventArgs) Handles ultraGridERPPurchaseOrders.AfterRowActivate
        Try
            If (IsGridLoading) Then
                Exit Sub
            End If

            Dim selectedRow As UltraGridRow = ultraGridERPPurchaseOrders.ActiveRow
            Dim PoLine As QuickReportSchema
            If (Not selectedRow.HasParent) Then
                PoLine = TryCast(selectedRow.ListObject, QuickReportSchema)
                Dim childBand = selectedRow.ChildBands(0)
                If childBand.Rows.Count > 0 Then
                    childBand.Rows(0).Activate()
                    Exit Sub
                End If
            Else
                Dim disassembly = TryCast(selectedRow.ListObject, DisassemblyPart)
                Dim parent = TryCast(selectedRow.ParentRow.ListObject, QuickReportSchema).Clone()

                If (parent.PartNo <> disassembly.ChildPartNo) Then
                    parent.PartNo = disassembly.ChildPartNo
                    parent.IFSInventory = parent.IFSInventory.Where(Function(item) item.PartNo = disassembly.ChildPartNo).ToList()
                    parent.InvQtyArrived = disassembly.Quantity
                    'parent.FinesseQty = 0
                End If

                If (parent IsNot Nothing AndAlso Not parent.IsSerializedPo) Then
                    txtQty.Text = disassembly.Quantity
                Else
                    txtQty.Text = 1
                End If
                PoLine = parent
            End If
            _selected_ifs_po = PoLine
            If (PoLine IsNot Nothing) Then
                cboRef4.SelectedValue = "PURCADD"
                If (Not String.IsNullOrEmpty(PoLine.Owner)) Then
                    cboOwner.SelectedValue = PoLine.Owner
                End If
                If (Not String.IsNullOrEmpty(PoLine.Warehouse)) Then
                    cboWarehouse.SelectedValue = PoLine.Warehouse
                End If

                txtPONumber.Text = PoLine.OrderNo
                txtPartNo.Text = PoLine.PartNo
                hideOrDisplayAutoIPAddressForPart(PoLine.PartNo)
                ReloadPermissions()
            End If

        Catch ex As Exception
            Console.WriteLine("Error Activating ultragrid row...")
            Console.WriteLine(ex.Message)
        End Try
    End Sub
