Private Function GetErpBaseData() As IEnumerable(Of QuickReportSchema)
        If (_ifs_data IsNot Nothing AndAlso cboErpWarehouse.SelectedItem IsNot Nothing) Then
            Dim selectedCompany = cboErpWarehouse.SelectedItem.Row("IFSCompanyCd").ToString
            Dim result = _ifs_data.Where(Function(po) _
                                        (String.IsNullOrEmpty(selectedCompany) OrElse String.IsNullOrEmpty(po.Contract) OrElse po.Contract = selectedCompany OrElse po.Warehouse = selectedCompany) AndAlso
    (String.IsNullOrEmpty(cboErpOwner.SelectedValue) OrElse String.IsNullOrEmpty(po.Owner) OrElse po.Owner = cboErpOwner.SelectedValue) AndAlso
                                        (chkUnexistingParts.Checked OrElse (Not chkUnexistingParts.Checked AndAlso (po.FinessePart IsNot Nothing OrElse (po.Disassemblies IsNot Nothing AndAlso po.Disassemblies.Count > 0)))) AndAlso
    (Not chkRecievedOnly.Checked OrElse
        (po.IsRecieved = True AndAlso
         CInt(po.FinesseQty) < CInt(po.BuyQtyDue) AndAlso 'Filtering by Qty ordered instead of qty arrived to show rows that haven't been fully receipt.
         (po.FinessePart IsNot Nothing OrElse po.Disassemblies IsNot Nothing)))
)



            Return result
        End If

        Return New List(Of QuickReportSchema)()
    End Function
